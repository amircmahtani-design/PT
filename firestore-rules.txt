rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Amir PT.
    // One document per Sync ID for your training data, plus <syncid>__cat0,
    // __cat1 ... for the exercise catalogue. The {docId} wildcard covers all
    // of them, so you never need to change this again.
    match /amirpt/{docId} {
      allow read, write: if request.auth != null;
    }

    // Nothing else in the project is readable or writable.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
